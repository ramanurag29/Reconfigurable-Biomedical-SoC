// ==== 04_RTL/neural_network/nn_engine.v
module nn_engine #(
    parameter ACC_W = 32
) (
    input  wire                     clk,
    input  wire                     rst_n,
    input  wire                     start,
    input  wire [4:0]               shift_cfg,     // S1 (hidden requantization shift)
    input  wire [15:0]              conf_thr,
    // external memories
    output wire [2:0]               feat_addr,
    input  wire signed [7:0]        feat_data,
    output wire [7:0]               w_addr,
    input  wire signed [7:0]        w_data,
    output wire [4:0]               b_addr,
    input  wire signed [23:0]       b_data,
    // shared-MAC port
    output wire                     mac_req,
    input  wire                     mac_grant,
    output wire                     mac_load,
    output wire signed [ACC_W-1:0]  mac_load_val,
    output wire                     mac_en,
    output wire signed [10:0]       mac_a,
    output wire signed [11:0]       mac_b,
    input  wire signed [ACC_W-1:0]  mac_acc,
    input  wire                     mac_acc_valid,
    input  wire                     mac_overflow,
    // result
    output reg                      nn_valid,
    output reg                      nn_class,
    output reg  signed [ACC_W-1:0]  nn_score0,
    output reg  signed [ACC_W-1:0]  nn_score1,
    output reg  [15:0]              nn_margin,
    output reg                      nn_lowconf,
    output reg                      nn_ovf,
    output wire                     busy,
    output wire                     at_boundary,   // 1 = safe preemption point (between MAC ops)
    output reg  [15:0]              last_latency
);
// ==== 04_RTL/scheduler/task_scheduler.v
module task_scheduler #(
    parameter ACC_W = 32,
    parameter A_W   = 11,
    parameter B_W   = 12,
    parameter PIPE  = 0
) (
    input  wire                     clk,
    input  wire                     rst_n,
    // configuration
    input  wire [1:0]               policy,
    input  wire                     mode_sel,       // mode-based: favoured client (0=FIR,1=NN)
    input  wire                     preempt_en,
    input  wire [2:0]               prio0, prio1,   // higher = more urgent
    input  wire [7:0]               quantum,        // RR slice in granted MAC ops (0 = never slice)
    input  wire [15:0]              deadline0, deadline1,   // cycles, 0 = no deadline monitoring
    input  wire                     stat_clr,
    // client 0 (FIR)
    input  wire                     req0, bnd0, clr0, load0, en0,
    input  wire signed [A_W-1:0]    a0,
    input  wire signed [B_W-1:0]    b0,
    input  wire signed [ACC_W-1:0]  lv0,
    output wire                     grant0,
    // client 1 (NN)
    input  wire                     req1, bnd1, clr1, load1, en1,
    input  wire signed [A_W-1:0]    a1,
    input  wire signed [B_W-1:0]    b1,
    input  wire signed [ACC_W-1:0]  lv1,
    output wire                     grant1,
    // results routed back to clients
    output wire signed [ACC_W-1:0]  acc_out,
    output wire                     acc_valid0, acc_valid1,
    output wire                     ovf0, ovf1,
    // status / waveform visibility
    output wire [1:0]               sched_state,
    output wire                     owner_o,
    // metrics
    output reg  [31:0]              m_total_cycles,
    output reg  [31:0]              m_mac_ops,       // client MAC ops issued (restore loads excluded)
    output reg  [15:0]              m_preempts,
    output reg  [31:0]              m_ovh_cycles,    // cycles lost in restore (scheduling overhead)
    output reg  [15:0]              m_tasks0, m_tasks1,
    output reg  [31:0]              m_wait_sum0, m_wait_sum1,
    output reg  [15:0]              m_wait_max0, m_wait_max1,
    output reg  [15:0]              m_lat_max0,  m_lat_max1,
    output reg  [31:0]              m_exec_sum0, m_exec_sum1,
    output reg  [15:0]              m_miss0, m_miss1
);
